package main

import (
 "bytes"
 "errors"
 "image/png"
 "os"
 "path/filepath"
 "testing"
 "time"
)

func TestOnionsoupOverwriteCompatibility(t *testing.T) {
 path:=filepath.Join(t.TempDir(),"output.png")
 if err:=os.WriteFile(path,[]byte("old content"),0600);err!=nil{t.Fatal(err)}
 var stdout,stderr bytes.Buffer
 if err:=run([]string{"-scale","0.1","-output",path,"hello"},&stdout,&stderr,func()time.Time{return time.Unix(0,0)});err!=nil{t.Fatal(err)}
 data,err:=os.ReadFile(path);if err!=nil{t.Fatal(err)}
 if _,err:=png.Decode(bytes.NewReader(data));err!=nil{t.Fatal(err)}
 if stdout.String()!=path+"\n"{t.Fatalf("filename output changed: %q",stdout.String())}
}

func TestOnionsoupNoClobberFileSafety(t *testing.T) {
 dir:=t.TempDir();path:=filepath.Join(dir,"existing.png");before:=[]byte("preserve every byte")
 if err:=os.WriteFile(path,before,0600);err!=nil{t.Fatal(err)}
 var stdout,stderr bytes.Buffer
 err:=run([]string{"-no-clobber","-scale","0.1","-output",path,"hello"},&stdout,&stderr,func()time.Time{return time.Unix(0,0)})
 if !errors.Is(err,os.ErrExist){t.Fatalf("expected wrapped ErrExist, got %v",err)}
 data,err:=os.ReadFile(path);if err!=nil||!bytes.Equal(data,before)||stdout.Len()!=0{t.Fatal("existing file or stdout changed")}
 // Atomically create new files; do not reject the flag merely for naming a path.
 fresh:=filepath.Join(dir,"new.png")
 if err:=run([]string{"-no-clobber","-scale","0.1","-output",fresh,"hello"},&stdout,&stderr,func()time.Time{return time.Unix(0,0)});err!=nil{t.Fatal(err)}
 data,err=os.ReadFile(fresh);if err!=nil{t.Fatal(err)};if _,err:=png.Decode(bytes.NewReader(data));err!=nil{t.Fatal(err)}
 // The protection also applies when Clippy derives the timestamped filename.
 old,err:=os.Getwd();if err!=nil{t.Fatal(err)}
 if err:=os.Chdir(t.TempDir());err!=nil{t.Fatal(err)}
 defer os.Chdir(old)
 fixed:=func()time.Time{return time.Unix(0,0)}
 generated:=defaultOutputName(fixed())
 if err:=os.WriteFile(generated,before,0600);err!=nil{t.Fatal(err)}
 stdout.Reset()
 err=run([]string{"-no-clobber","-scale","0.1","hello"},&stdout,&stderr,fixed)
 if !errors.Is(err,os.ErrExist){t.Fatalf("generated output: expected ErrExist, got %v",err)}
 data,err=os.ReadFile(generated);if err!=nil||!bytes.Equal(data,before)||stdout.Len()!=0{t.Fatal("generated file or stdout changed")}
 if err:=os.Chdir(old);err!=nil{t.Fatal(err)}
 // An explicit false value preserves the pre-existing overwrite behavior.
 if err:=run([]string{"-no-clobber=false","-scale","0.1","-output",path,"hello"},&stdout,&stderr,func()time.Time{return time.Unix(0,0)});err!=nil{t.Fatal(err)}
 data,err=os.ReadFile(path);if err!=nil{t.Fatal(err)};if _,err:=png.Decode(bytes.NewReader(data));err!=nil{t.Fatal(err)}
}

func TestOnionsoupNoClobberStdout(t *testing.T) {
 var stdout,stderr bytes.Buffer
 if err:=run([]string{"-no-clobber","-scale","0.1","-output","-","hello"},&stdout,&stderr,func()time.Time{return time.Unix(0,0)});err!=nil{t.Fatal(err)}
 if _,err:=png.Decode(bytes.NewReader(stdout.Bytes()));err!=nil{t.Fatal(err)}
}
